-- Normalize questions when saving and matching so capitalization and punctuation do not matter.
function normalizeRiddleQuestion(value)
    local normalized = tostring(value or ""):lower()
    normalized = normalized:gsub("[%p]", " ")
    normalized = normalized:gsub("%s+", " ")
    return normalized:match("^%s*(.-)%s*$") or ""
end

function trimRiddleAnswer(value)
    return tostring(value or ""):match("^%s*(.-)%s*$") or ""
end

_useRiddleDB = savedConfig.useRiddleDB
-- When ON, riddles the database can't answer are sent to Gemini AI.
_useAI = savedConfig.useAI

function answerRiddle(text)
    if not _riddleSolver or not _useRiddleDB or type(text) ~= "string" then return nil end

    local normalizedText = normalizeRiddleQuestion(text)
    if normalizedText == "" then return nil end

    local bestAnswer = nil
    local bestLength = 0
    for question, answer in pairs(riddleDB) do
        local normalizedQuestion = normalizeRiddleQuestion(question)
        if normalizedQuestion ~= "" and normalizedText:find(normalizedQuestion, 1, true) then
            if #normalizedQuestion > bestLength then
                bestAnswer = answer
                bestLength = #normalizedQuestion
            end
        end
    end
    return bestAnswer
end

function isOurGui(instance)
    local p = instance
    for _ = 1, 10 do
        if not p then break end
        if p.Name == "CALAMITY SNIPER" or p.Name == "ROSE.VS SNIPER" or p.Name == "ACECodeSniperUI" or p.Name == "SourcesHubRedeemerGui" then return true end
        p = p.Parent
    end
    return false
end

function isVisibleChain(inst)
    local current = inst
    while current do
        if current:IsA("GuiObject") and not current.Visible then return false end
        if current:IsA("ScreenGui") then return current.Enabled end
        current = current.Parent
    end
    return true
end

function findAllTextBoxes(pg)
    local boxes = {}
    for _, gui in ipairs(pg:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Enabled and not isOurGui(gui) then
            for _, d in ipairs(gui:GetDescendants()) do
                if d:IsA("TextBox") and not isOurGui(d) then
                    boxes[#boxes+1] = d
                end
            end
        end
    end
    return boxes
end

function findCodeButtons(pg)
    local btns = {}
    for _, gui in ipairs(pg:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Enabled and not isOurGui(gui) then
            for _, d in ipairs(gui:GetDescendants()) do
                if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) then
                    local n  = d.Name:lower()
                    local pn = (d.Parent and d.Parent.Name or ""):lower()
                    if (n:find("code") or n:find("redeem") or pn:find("code") or pn:find("redeem"))
                        and isVisibleChain(d) then
                        btns[#btns+1] = d
                    end
                end
            end
        end
    end
    return btns
end

function clickButton(btn)
    if not btn then return false end
    local methods = {}
    
    methods[#methods+1] = function() btn.MouseButton1Click:Fire() end
    methods[#methods+1] = function() btn.Activated:Fire() end
    
    if typeof(firesignal) == "function" then
        methods[#methods+1] = function() firesignal(btn.MouseButton1Click) end
        methods[#methods+1] = function() firesignal(btn.Activated) end
    end
    
    if typeof(getconns) == "function" then
        methods[#methods+1] = function()
            local ok, cs = pcall(getconns, btn.MouseButton1Click)
            if ok and type(cs) == "table" then
                for _, c in ipairs(cs) do pcall(function() c:Fire() end) end
            end
            local ok2, cs2 = pcall(getconns, btn.Activated)
            if ok2 and type(cs2) == "table" then
                for _, c in ipairs(cs2) do pcall(function() c:Fire() end) end
            end
        end
    end
    
    if typeof(fireclick) == "function" then
        methods[#methods+1] = function() fireclick(btn) end
    end
    
    local anyOk = false
    for _, fn in ipairs(methods) do
        local ok = pcall(fn)
        anyOk = anyOk or ok
    end
    return anyOk
end

function fireBoxFocusLost(box)
    if not box then return false end
    local anyFired = false
    
    if typeof(firesignal) == "function" then
        local ok = pcall(firesignal, box.FocusLost, true)
        anyFired = anyFired or ok
    end
    
    if typeof(getconns) == "function" then
        local ok, cs = pcall(getconns, box.FocusLost)
        if ok and type(cs) == "table" then
            for _, c in ipairs(cs) do
                local fn
                pcall(function() fn = c.Function end)
                if fn and typeof(getupvalues) == "function" and typeof(setupv) == "function" then
                    local uOk, ups = pcall(getupvalues, fn)
                    if uOk and type(ups) == "table" then
                        for i, v in pairs(ups) do
                            if type(v) == "boolean" and v == true then
                                pcall(setupv, fn, i, false)
                            end
                        end
                    end
                end
                local fOk = pcall(function()
                    if c.Enabled ~= false then c:Fire(true) end
                end)
                anyFired = anyFired or fOk
            end
        end
    end
    
    return anyFired
end

function typeAndSubmitCode(code)
    local pg = playerGui or player:FindFirstChildOfClass("PlayerGui")
    if not pg then return false, "no PlayerGui" end

    -- Strategy 1: Known UI path
    local codesGui = pg:FindFirstChild("Codes")
    if codesGui then
        if codesGui:IsA("ScreenGui") then codesGui.Enabled = true end
        local codesFrame = codesGui:FindFirstChild("Codes") or codesGui
        if codesFrame then
            if codesFrame:IsA("GuiObject") then codesFrame.Visible = true end
            local cur = codesFrame
            while cur and cur ~= codesGui do
                if cur:IsA("GuiObject") then cur.Visible = true end
                cur = cur.Parent
            end

            local box = nil
            for _, d in ipairs(codesFrame:GetDescendants()) do
                if d:IsA("TextBox") and not isOurGui(d) then
                    box = d
                    break
                end
            end

            local submitBtn = nil
            for _, d in ipairs(codesFrame:GetDescendants()) do
                if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) then
                    local n = d.Name:lower()
                    local txt = ""
                    pcall(function() txt = d.Text:lower() end)
                    if n:find("submit") or txt:find("submit") or n:find("redeem") or txt:find("redeem") or n:find("claim") or txt:find("confirm") or n:find("enter") then
                        submitBtn = d
                        break
                    end
                end
            end
            if not submitBtn then
                for _, d in ipairs(codesFrame:GetDescendants()) do
                    if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) then
                        local n = d.Name:lower()
                        if not n:find("close") and not n:find("x") and not n:find("toggle") then
                            submitBtn = d
                            break
                        end
                    end
                end
            end

            if box then
                pcall(function() box.Text = code end)
                task.wait()
                if submitBtn then clickButton(submitBtn) end
                fireBoxFocusLost(box)
                return true, "submitted via PlayerGui.Codes"
            end
        end
    end

    -- Strategy 2: Dynamic Search (single-frame steps, short deadline)
    local btns = findCodeButtons(pg)
    for _, btn in ipairs(btns) do
        clickButton(btn)
    end

    task.wait()

    local box = nil
    local deadline = tick() + 0.8
    while tick() < deadline do
        local allBoxes = findAllTextBoxes(pg)
        for _, d in ipairs(allBoxes) do
            if isVisibleChain(d) then
                local n  = d.Name:lower()
                local pn = (d.Parent and d.Parent.Name or ""):lower()
                if n:find("code") or pn:find("code") or n:find("redeem") or pn:find("redeem") or n:find("input") or pn:find("textbox") or n:find("enter") then
                    box = d
                    break
                end
            end
        end
        if not box then
            for _, d in ipairs(allBoxes) do
                if isVisibleChain(d) then box = d; break end
            end
        end
        if box then break end
        task.wait()
    end

    if not box then return false, "no codebox visible" end

    pcall(function() box.Text = code end)
    task.wait()

    local redeemBtn = nil
    local searchNames = {"submit","redeem","claim","confirm","enter","send","apply","ok","use","go","check"}
    local p = box.Parent
    for _ = 1, 8 do
        if not p then break end
        for _, d in ipairs(p:GetDescendants()) do
            if (d:IsA("TextButton") or d:IsA("ImageButton")) and not isOurGui(d) and d ~= box then
                local n = d.Name:lower()
                local txt = ""
                pcall(function() txt = d.Text:lower() end)
                for _, sn in ipairs(searchNames) do
                    if n:find(sn) or txt:find(sn) then
                        if isVisibleChain(d) then
                            redeemBtn = d
                            break
                        end
                    end
                end
                if redeemBtn then break end
            end
        end
        if redeemBtn then break end
        p = p.Parent
    end

    if redeemBtn then clickButton(redeemBtn) end
    fireBoxFocusLost(box)

    return true, "submitted via dynamic search"
end

-- Try the direct image first. If Roblox cannot serve it, keep the UI readable with
-- a built-in gradient/text fallback rather than showing an invisible blank panel.
function themeAsset(assetId)
    local id = tostring(assetId or ""):gsub("rbxassetid://", "")
    return "rbxassetid://" .. id
end
function themeThumbnail(assetId)
    local id = tostring(assetId or ""):gsub("rbxassetid://", "")
    return "rbxthumb://type=Asset&id=" .. id .. "&w=420&h=420"
end
-- Some executors/clients refuse to render a raw rbxassetid for DECAL assets (they
-- only like true image assets) and silently show nothing. The rbxthumb endpoint
-- resolves ANY public decal/image, so it loads FIRST; the direct asset id is only
-- used as a backup if the thumbnail service hiccups. Cosmetic only.
function loadThemeImage(imageObject, assetId)
    if not imageObject then return end
    imageObject.Image = themeThumbnail(assetId)
    imageObject.ImageTransparency = 0
    task.spawn(function()
        local ContentProvider = game:GetService("ContentProvider")
        local ok = pcall(function() ContentProvider:PreloadAsync({imageObject}) end)
        if not ok and imageObject and imageObject.Parent then
            imageObject.Image = themeAsset(assetId)
            pcall(function() ContentProvider:PreloadAsync({imageObject}) end)
        end
    end)
end

-- Toggle controls register themselves here so their ON/OFF colors are painted

    accentObjects[#accentObjects + 1] = { obj = obj, get = getState, stroke = stroke, knob = knob }
end

function applyAccentColors()
    for _, e in ipairs(accentObjects) do
        local on = (e.get and e.get()) or false
        if e.obj then e.obj.BackgroundColor3 = on and COLORS.Accent or COLORS.Control end
        if e.stroke then e.stroke.Transparency = on and 0.62 or 0.88 end
        if e.knob then e.knob.BackgroundColor3 = on and COLORS.Window or COLORS.White end
    end
end

function addStroke(parent, color, thickness, transparency)
    local value = Instance.new("UIStroke")
    value.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    -- Fixed theme: accent strokes are painted straight away, no recolor registry.
    if color == COLORS.White or color == COLORS.Accent then
        color = THEME.accentBright
    end
    value.Color = color
    value.Thickness = thickness or 1
    value.Transparency = transparency or 0
    value.Parent = parent
    return value
end

-- ==========================================================
-- ANIMATED GRADIENT STROKES
-- One UIGradient per border; a single RenderStepped loop rotates them all, so
-- the whole window costs a handful of property writes per frame. The gradient
-- carries both the colour band and a transparency band, which keeps the base
-- hairline exactly as quiet as it was and only brightens the travelling glint.
-- ==========================================================
STROKE_FX = {
    speed   = 46,                 -- degrees / second
    lift    = 0.30,               -- how far the band lightens the hairline colour
    punch   = 0.06,               -- band transparency, as a fraction of the resting one
    enabled = savedConfig.strokeFx ~= false,
    entries = {},                 -- stroke -> entry (also used as a skip set)
}

-- Right-to-left sweep for the Discord wordmark. Declared here, with the other
-- FX settings, because setStrokeFx() parks it and runs long before the label
-- itself is built.
DISCORD_FX = {
    speed = 0.32,                 -- full sweeps per second (~3s per pass); negative reverses it
    offset = 0.5,
}
-- Band shape: quiet border, ramp, bright core, ramp, quiet border.
local FX_BAND = { 0.00, 0.28, 0.44, 0.50, 0.56, 0.72, 1.00 }
local function fxKeypoints(kind, values)
    local out = {}
    for i = 1, #FX_BAND do
        out[i] = kind.new(FX_BAND[i], values[i])
    end
    return out
end
auraStrokeGradients = {}
local auraGradientAnimation

local function mixColor3(a, b, t)
    return Color3.new(
        a.R + (b.R - a.R) * t,
        a.G + (b.G - a.G) * t,
        a.B + (b.B - a.B) * t
    )
end

-- Gradient sweep built from the fixed theme palette (window outline).
function auraStrokeSequence()
    local t = THEME
    local mid = mixColor3(t.accentBright, t.accent, 0.55)
    return ColorSequence.new({
        ColorSequenceKeypoint.new(0.00, t.accentBright),
        ColorSequenceKeypoint.new(0.25, mid),
        ColorSequenceKeypoint.new(0.50, t.deep),
        ColorSequenceKeypoint.new(0.75, mid),
        ColorSequenceKeypoint.new(1.00, t.accentBright),
    })
end

-- hairline -> light -> hairline: reads as a highlight glinting along the edge.
function glowStrokeSequence(base)
    local top = mixColor3(base, Color3.new(1, 1, 1), STROKE_FX.lift)
    local near = mixColor3(base, top, 0.45)
    return ColorSequence.new(fxKeypoints(ColorSequenceKeypoint,
        { base, base, near, top, near, base, base }))
end

function glowStrokeTransparency(floor)
    local peak = math.max(0, floor * STROKE_FX.punch)
    local near = floor * 0.35 + peak * 0.65
    return NumberSequence.new(fxKeypoints(NumberSequenceKeypoint,
        { floor, floor, near, peak, near, floor, floor }))
end

function paintGlowStroke(entry)
    if not entry.stroke or not entry.stroke.Parent then return end
    if not entry.gradient then
        local gradient = Instance.new("UIGradient")
        gradient.Name = "AuraAnimatedGradient"
        gradient.Color = entry.sequence and entry.sequence() or glowStrokeSequence(entry.base)
        gradient.Transparency = glowStrokeTransparency(entry.floor)
        gradient.Rotation = entry.phase
        gradient.Parent = entry.stroke
        entry.gradient = gradient
    end
    entry.stroke.Transparency = 0
end

function unpaintGlowStroke(entry)
    if entry.gradient then
        pcall(function() entry.gradient:Destroy() end)
        entry.gradient = nil
    end
    if entry.stroke and entry.stroke.Parent then
        entry.stroke.Transparency = entry.floor
    end
end

function registerGlowStroke(stroke, speed, sequenceFn)
    if not stroke or STROKE_FX.entries[stroke] then return stroke end
    local entry = {
        stroke = stroke,
        base = stroke.Color,
        floor = stroke.Transparency or 0,
        phase = (#auraStrokeGradients * 41) % 360,
        speed = speed or 1,
        sequence = sequenceFn,
        gradient = nil,
    }
    STROKE_FX.entries[stroke] = entry
    auraStrokeGradients[#auraStrokeGradients + 1] = entry
    if STROKE_FX.enabled then paintGlowStroke(entry) end
    return stroke
end

-- Flip the whole window between animated and plain hairline borders.
function setStrokeFx(on)
    STROKE_FX.enabled = on and true or false
    -- Switched off: park the wordmark mid-ramp instead of freezing it wherever
    -- the sweep happened to stop.
    if not STROKE_FX.enabled then
        DISCORD_FX.offset = 0
        if DiscordGradient then pcall(function() DiscordGradient.Offset = Vector2.new(0, 0) end) end
    end
    for index = #auraStrokeGradients, 1, -1 do
        local entry = auraStrokeGradients[index]
        if not entry.stroke or not entry.stroke.Parent then
            STROKE_FX.entries[entry.stroke] = nil
            table.remove(auraStrokeGradients, index)
        elseif STROKE_FX.enabled then
            paintGlowStroke(entry)
        else
            unpaintGlowStroke(entry)
        end
    end
    return STROKE_FX.enabled
end

-- Legacy entry point: the window frame keeps its theme-palette sweep.
function addAnimatedGradientStroke(parent, thickness, transparency, phase)
    local stroke = addStroke(parent, COLORS.White, thickness or 1.5, transparency or 0.06)
    registerGlowStroke(stroke, 1, auraStrokeSequence)
    local entry = STROKE_FX.entries[stroke]
    if entry then
        entry.phase = phase or 0
    elseif stroke then
        STROKE_FX.entries[stroke] = nil
    end
    return stroke
end

-- Hover / press feedback for every clickable control (pcall-guarded: a stripped
-- executor without TweenService just keeps the flat look instead of erroring).
function addButtonFX(btn, baseTransparency, hoverLift)
    if not btn then return end
    baseTransparency = baseTransparency or 0
    hoverLift = hoverLift or 0.16
    local hovering = false
    local function tintTo(v, dur)
        pcall(function()
            TweenService:Create(btn, TweenInfo.new(dur or 0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                { BackgroundTransparency = v }):Play()
        end)
    end
    btn.MouseEnter:Connect(function() hovering = true; tintTo(math.max(0, baseTransparency - hoverLift)) end)
    btn.MouseLeave:Connect(function() hovering = false; tintTo(baseTransparency) end)
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tintTo(math.max(0, baseTransparency - hoverLift - 0.10), 0.06)
        end
    end)
    btn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tintTo(hovering and math.max(0, baseTransparency - hoverLift) or baseTransparency, 0.10)
        end
    end)
end

-- Animate every registered border from one RenderStepped connection.
auraGradientAnimation = RunService.RenderStepped:Connect(function()
    if not GUI or not GUI.Parent then
        if auraGradientAnimation then
            auraGradientAnimation:Disconnect()
            auraGradientAnimation = nil
        end
        return
    end
    -- The wordmark is a child of GUI, not the panel, so it keeps sweeping while
    -- the window is minimized. It is driven here instead of in its own
    -- connection so the whole GUI stays on one RenderStepped heartbeat.
    local wordmark = DiscordGradient
    if STROKE_FX.enabled and wordmark and wordmark.Parent then
        local clock = os.clock()
        -- decreasing offset = the ramp travels right to left; flip the sign to
        -- reverse it. Kept clock-derived (not accumulated) so a hitch cannot
        -- make the sweep drift or jump.
        local phase = 0.5 - ((clock * DISCORD_FX.speed) % 1)
        if phase < -0.5 then phase = phase + 1 end
        DISCORD_FX.offset = phase
        wordmark.Offset = Vector2.new(phase, 0)
    end

    -- no point sweeping borders nobody can see
    if not STROKE_FX.enabled or windowMinimized then return end

    local now = os.clock()
    for index = #auraStrokeGradients, 1, -1 do
        local entry = auraStrokeGradients[index]
        local stroke = entry.stroke
        if not stroke or not stroke.Parent then
            STROKE_FX.entries[stroke] = nil
            table.remove(auraStrokeGradients, index)
        else
            local gradient = entry.gradient
            if gradient and gradient.Parent then
                gradient.Rotation = (now * STROKE_FX.speed * (entry.speed or 1) + entry.phase) % 360
            end
        end
    end
end)

        local bottom = math.max(0, Console.AbsoluteCanvasSize.Y - Console.AbsoluteWindowSize.Y)
        Console.CanvasPosition = Vector2.new(0, bottom)
    end)
end

-- Escapes free-form text for the RichText console.
local function richEscape(text)
    return tostring(text):gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")
end

-- Free-form console line, same grammar as the setting lines. Used by the AI
-- self-test so a result outlives the status ticker (and shows even when the
-- master switch is off).
-- The ACTIVITY box now has exactly ONE writer. Two writers meant one event
-- could land twice (the AI test logged every line once untagged and once as
-- "[ai-test]"), and nothing ever trimmed the box, so a long session kept
-- repainting hundreds of lines.
CONSOLE_MAX_LINES = 45

function appendConsoleLine(line)
    if not ConsoleOutput then return false end
    local text = ConsoleOutput.Text
    if text == "" then ConsoleOutput.Text = line
    else ConsoleOutput.Text = text .. "\n\n" .. line end
    local parts = {}
    for chunk in ConsoleOutput.Text:gmatch("[^\n]+") do
        if chunk ~= "" then parts[#parts + 1] = chunk end
    end
    if #parts > CONSOLE_MAX_LINES then
        local kept = {}
        for i = #parts - CONSOLE_MAX_LINES + 1, #parts do kept[#kept + 1] = parts[i] end
        ConsoleOutput.Text = table.concat(kept, "\n\n")
    end
    scrollConsoleToBottom()
    return true
end

function consoleNote(tag, text, col)
    appendConsoleLine('<font color="' .. CONSOLE_COLORS.Dim .. '">[' .. tag .. ']</font> '
        .. '<font color="' .. col3ToRich(col or COLORS.Dim) .. '">' .. richEscape(text) .. "</font>")
end

function appendConsoleStatus(name, activated)
    if not ConsoleOutput then return end
    local state = activated and "ON" or "OFF"
    local stateColor = activated and CONSOLE_COLORS.Green or CONSOLE_COLORS.Red
    local line = '<font color="' .. CONSOLE_COLORS.Dim .. '">[setting]</font> '
        .. '<font color="' .. CONSOLE_COLORS.Amber .. '">' .. name .. "</font> "
        .. '<font color="' .. CONSOLE_COLORS.Dim .. '">-&gt;</font> '
        .. '<font color="' .. stateColor .. '">' .. state .. "</font>"
    appendConsoleLine(line)
end

function makeStateButton(parent, enabled, consoleName, onToggle)
    parent.Active = true
    featureStates[consoleName] = enabled
    local track, knob, paint = makeSwitch(parent, "State", enabled, (parent.ZIndex or 1) + 1)
    track.Position = UDim2.new(1, -50, 0.5, -10)
    local state = enabled
    local lastSubToggle = 0

    local function toggleState()
        if tick() - lastSubToggle < 0.15 then return end
        lastSubToggle = tick()
        state = not state
        featureStates[consoleName] = state
        paint(state)
        if autoWriteEnabled then appendConsoleStatus(consoleName, state) end
        if onToggle then onToggle(state) end
    end

    track.Activated:Connect(toggleState)
    track.MouseButton1Click:Connect(toggleState)
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            toggleState()
        end
    end)
    registerAccentObject(track, function() return state end, nil, nil)
    return track
end

-- Adding the same normalized question updates its answer instead of creating a duplicate.
function addRiddleEntry()
    local question = normalizeRiddleQuestion(newQuestionBox.Text)
    local answer = trimRiddleAnswer(newAnswerBox.Text)

    if question == "" then
        riddleAddStatus.Text = "QUESTION REQUIRED"
        riddleAddStatus.TextColor3 = COLORS.Red
        newQuestionBox:CaptureFocus()
        return
    end
    if answer == "" then
        riddleAddStatus.Text = "ANSWER REQUIRED"
        riddleAddStatus.TextColor3 = COLORS.Red
        newAnswerBox:CaptureFocus()
        return
    end

    local updating = riddleDB[question] ~= nil
    riddleDB[question] = answer
    saveRiddleDB()
    populateRiddleList()

    newQuestionBox.Text = ""
    newAnswerBox.Text = ""
    riddleAddStatus.Text = updating and "ANSWER UPDATED" or "RIDDLE ADDED"
    riddleAddStatus.TextColor3 = COLORS.Green
    setStatus((updating and "Riddle updated: " or "Riddle added: ") .. question, COLORS.Green)
end

-- Wire the travelling highlight into every structural border. Rows live in lists
-- that rebuild hundreds of times, so anything under those two hosts is skipped,
-- and so are strokes whose transparency already spells ON / OFF.
do
    local skipHosts = {}
    for _, host in ipairs({ RiddleList, toolList }) do
        if host then skipHosts[host] = true end
    end
    for _, accent in ipairs(accentObjects) do
        if accent.stroke then STROKE_FX.entries[accent.stroke] = { stroke = accent.stroke } end
    end
    local function inList(node)
        while node and node ~= Window do
            if skipHosts[node] then return true end
            node = node.Parent
        end
        return false
    end
    for _, decendant in ipairs(Window:GetDescendants()) do
        if decendant:IsA("UIStroke") and not STROKE_FX.entries[decendant]
            and not inList(decendant.Parent) then
            registerGlowStroke(decendant)
        end
    end
    setStrokeFx(savedConfig.strokeFx ~= false)
    if type(warn) == "function" then
        warn("[CALAMITY SNIPER] animated gradient borders: " .. #auraStrokeGradients
            .. (STROKE_FX.enabled and "" or " (Stroke glow is OFF)"))
    end
end

-- WINDOW DRAGGING SYSTEM WITH TOUCH INTEGRATION --
do
    dragging = false
    local activeDragInput
    local dragStart
    local startPosition
    dragMoved = false
    DRAG_THRESHOLD = UserInputService.TouchEnabled and 10 or 3

    function isOverButton(button, position)
        local btnPos = button.AbsolutePosition
        local btnSize = button.AbsoluteSize
        return position.X >= (btnPos.X - 10)
            and position.X <= (btnPos.X + btnSize.X + 10)
            and position.Y >= (btnPos.Y - 10)
            and position.Y <= (btnPos.Y + btnSize.Y + 10)
    end

    function isOverHeaderControl(position)
        return isOverButton(AutoWriteButton, position) or isOverButton(MinimizeButton, position)
    end

    function stopDragging(input)
        if input ~= activeDragInput then return end
        dragging = false
        activeDragInput = nil
        dragStart = nil
        startPosition = nil
    end

    Header.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if dragging or isOverHeaderControl(input.Position) then return end

        dragging = true
        activeDragInput = input
        dragStart = Vector2.new(input.Position.X, input.Position.Y)
        startPosition = Window.Position
        dragMoved = false

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End or input.UserInputState == Enum.UserInputState.Cancel then
                stopDragging(input)
            end
        end)
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging or not activeDragInput then return end
        local isTrackedTouch = activeDragInput.UserInputType == Enum.UserInputType.Touch and input == activeDragInput
        local isTrackedMouse = activeDragInput.UserInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseMovement
        if not isTrackedTouch and not isTrackedMouse then return end

        local current = Vector2.new(input.Position.X, input.Position.Y)
        local delta = current - dragStart
        if not dragMoved then
            if delta.Magnitude < DRAG_THRESHOLD then return end
            dragMoved = true
        end

        Window.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end)
end

function col3ToRich(col)
    if col == COLORS.Green then return CONSOLE_COLORS.Green end
    if col == COLORS.Red then return CONSOLE_COLORS.Red end
    if col == COLORS.Text then return CONSOLE_COLORS.Amber end
    if col == COLORS.White then return CONSOLE_COLORS.Cyan end
    if col == COLORS.Dim then return CONSOLE_COLORS.Dim end
    return string.format("rgb(%d,%d,%d)", math.floor(col.R * 255 + 0.5), math.floor(col.G * 255 + 0.5), math.floor(col.B * 255 + 0.5))
end

-- statusTick drives only the bottom ticker; setStatus also logs. The AI test
-- needs the ticker without duplicating its tagged log line.
function statusTick(msg, col)
    if StatusTicker then
        StatusTicker.Text = tostring(msg)
        StatusTicker.TextTransparency = 0.65
        pcall(function()
            TweenService:Create(StatusTicker, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                { TextTransparency = 0 }):Play()
        end)
    end
end

function setStatus(msg, col)
    statusTick(msg, col)
    if not _enabled then return end
    if msg == _lastStatusMsg then return end
    _lastStatusMsg = msg
    col = col or COLORS.Dim
    appendConsoleLine('<font color="' .. col3ToRich(col) .. '">' .. tostring(msg) .. "</font>")
end

function flashCode(code, col)
    if not code or code == "" or code == "—" then return end
    setStatus("[code] -> " .. tostring(code), col or COLORS.White)
end

function resetPasteCounter() _capturedParts = {} end
clearAceCapture = function() _capturedParts = {} end

function clearBoxWatchers()
    if _boxTextConn then pcall(function() _boxTextConn:Disconnect() end) end
    if _boxAncestryConn then pcall(function() _boxAncestryConn:Disconnect() end) end
    for _, connection in ipairs(_boxVisibilityConns) do pcall(function() connection:Disconnect() end) end
    _boxTextConn = nil
    _boxAncestryConn = nil
    _boxVisibilityConns = {}
    _lastWatchedBox = nil
end

function watchBoxForBlankReset(box)
    if not box or _lastWatchedBox == box then return end
    clearBoxWatchers()
    _lastWatchedBox = box
    if box.Text ~= "" then _lastNonBlankBoxText = box.Text end
    _boxTextConn = box:GetPropertyChangedSignal("Text"):Connect(function()
        if box.Text == "" then resetPasteCounter()
        else _lastNonBlankBoxText = box.Text end
    end)
    _boxAncestryConn = box.AncestryChanged:Connect(function(_, parent)
        if not parent then resetPasteCounter(); clearBoxWatchers() end
    end)
end

UserInputService.TextBoxFocused:Connect(function(box)
    if box:IsDescendantOf(GUI) then return end
    if box ~= aceCodeBox() then return end
    _focused = box
    _lastBox = box
    watchBoxForBlankReset(box)
    if _enabled then setStatus("Ready", COLORS.Green) end
end)

UserInputService.TextBoxFocusReleased:Connect(function(box)
    if box:IsDescendantOf(GUI) then return end
    local codeBox = aceCodeBox()
    if box ~= codeBox and box ~= _lastBox then return end
    if _retypeInvalid and rememberPendingSubmission and (box == codeBox or box == _lastBox) then
        local submittedText = box.Text ~= "" and box.Text or _lastNonBlankBoxText
        rememberPendingSubmission(box, submittedText, false)
    end
    if _focused == box then
        _focused = nil
        if _enabled then
            setStatus((_lastBox and _lastBox.Parent) and "Ready" or "Click code box first", (_lastBox and _lastBox.Parent) and COLORS.Green or COLORS.Dim)
        end
    end
end)

clearPendingSubmission = function()
    _pendingRejectedToken += 1
    _pendingRejectedText = nil
    _pendingRejectedBox = nil
    _pendingRejectedUntil = 0
end

rememberPendingSubmission = function(box, text, replaceExisting)
    if not _retypeInvalid or not text or text == "" then return end
    if not replaceExisting and _pendingRejectedText and os.clock() <= _pendingRejectedUntil then return end
    _pendingRejectedToken += 1
    local token = _pendingRejectedToken
    _pendingRejectedText = text
    _pendingRejectedBox = box
    _pendingRejectedUntil = os.clock() + 8
    task.delay(8, function()
        if token == _pendingRejectedToken then clearPendingSubmission() end
    end)
end

function restoreRejectedText(box, previousText)
    if not _retypeInvalid or not previousText or previousText == "" then return false end
    RunService.Heartbeat:Wait()
    local repasteBox = aceCodeBox() or box
    if not repasteBox or not isVisibleChain(repasteBox) then return false end
    local restored = pcall(function() repasteBox.Text = previousText end)
    if restored then
        _lastBox = repasteBox
        watchBoxForBlankReset(repasteBox)
    end
    return restored
end

handleRedemptionFeedback = function(text, feedbackObject)
    if not _retypeInvalid or not _pendingRejectedText then return end
    if os.clock() > _pendingRejectedUntil then clearPendingSubmission(); return end
    if feedbackObject and feedbackObject:IsDescendantOf(GUI) then return end
    local lower = tostring(text or ""):lower()
    local rejected = lower:find("invalid code", 1, true)
        or lower:find("code is invalid", 1, true)
        or lower:find("expired", 1, true)
        or lower:find("already redeemed", 1, true)
        or lower:find("already used", 1, true)
        or lower:find("doesn't exist", 1, true)
        or lower:find("does not exist", 1, true)
        or lower:find("not found", 1, true)
        or lower:find("rejected", 1, true)
    if not rejected then return end
    local previousText = _pendingRejectedText
    local previousBox = _pendingRejectedBox
    local restored = restoreRejectedText(previousBox, previousText)
    clearPendingSubmission()
    if restored then
        setStatus("Invalid - repasted: " .. previousText, COLORS.Text)
        flashCode(previousText, COLORS.Red)
    end
end

function appendToBox(text)
    if not text or text == "" then return end
    if _lastWatchedBox and not isVisibleChain(_lastWatchedBox) then
        resetPasteCounter()
        clearBoxWatchers()
    end
    local box = aceCodeBox()
    _capturedParts[#_capturedParts + 1] = text
    local combinedCode = table.concat(_capturedParts)
    local capturedCount = #_capturedParts

    if box then
        _lastBox = box
        watchBoxForBlankReset(box)
        local boxWasFocused = UserInputService:GetFocusedTextBox() == box
        box.Text = combinedCode
        if boxWasFocused then
            pcall(function()
                local caretEnd = #combinedCode + 1
                box.CursorPosition = caretEnd
                box.SelectionStart = caretEnd
            end)
        end
    else
        setStatus("Captured; opening & searching UI...", COLORS.Text)
    end

    setStatus("Pasted " .. tostring(capturedCount) .. "/" .. tostring(_submitAfter), COLORS.Green)
    flashCode(combinedCode, COLORS.Green)

    if capturedCount >= _submitAfter then
        _capturedParts = {}
        if _autoAccept then
            rememberPendingSubmission(box, combinedCode, true)
            
            local ok, statusMsg = typeAndSubmitCode(combinedCode)
            
            if ok then
                setStatus("Redeemed: " .. combinedCode, COLORS.Green)
            else
                local restored = restoreRejectedText(box, combinedCode)
                clearPendingSubmission()
                if restored then
                    setStatus("Invalid - repasted: " .. combinedCode, COLORS.Text)
                    flashCode(combinedCode, COLORS.Red)
                else
                    setStatus("Failed: " .. tostring(statusMsg), COLORS.Red)
                end
            end
        end
    end
end

-- Riddle answers submit immediately on their own, never mixed with sniped code parts
-- or gated behind the "Submit after msgs" counter. Global on purpose: the main chunk
-- is at Luau's 200-local limit.
function submitRiddleAnswer(answer, label)
    if not answer or answer == "" then return end
    _capturedParts = {}
    local box = aceCodeBox()
    if box then
        _lastBox = box
        pcall(function() box.Text = answer end)
    end
    if not _autoAccept then
        setStatus((label or "Riddle") .. " answer pasted (auto submit OFF): " .. answer, COLORS.Text)
        return
    end
    rememberPendingSubmission(box, answer, true)
    local ok, statusMsg = typeAndSubmitCode(answer)
    if ok then
        setStatus((label or "Riddle") .. " answer submitted: " .. answer, COLORS.Green)
    else
        clearPendingSubmission()
        setStatus((label or "Riddle") .. " submit failed: " .. tostring(statusMsg), COLORS.Red)
    end
end

-- ANNOUNCEMENT NOTIFICATION LISTENER --
-- Looks through RemoteEvents in `root` for the one the game's notification
-- controller listens on. Needs the introspection helpers, so report when they are
-- missing instead of silently returning nil.
function scanRootForNotifier(root)
    local getinfo = debug and (debug.getinfo or debug.info)
    if not (getgc and getconnections and getinfo) then
        notifyScanNote = "this executor has no getgc/getconnections/debug.getinfo, so the game's notify remote cannot be located"
        return nil
    end
    if not root then return nil end
    local needles = { "NotificationController", "Notification", "Notify", "Announce" }
    local candidates = {}
    if root:IsA("RemoteEvent") then candidates[#candidates + 1] = root end
    local okDesc, descendants = pcall(function() return root:GetDescendants() end)
    if okDesc and type(descendants) == "table" then
        for _, d in ipairs(descendants) do
            if d:IsA("RemoteEvent") then candidates[#candidates + 1] = d end
        end
    end
    local best = nil
    for _, d in ipairs(candidates) do
        local ok, cs = pcall(getconnections, d.OnClientEvent)
        if ok and type(cs) == "table" then
            for _, c in ipairs(cs) do
                local f, fn = pcall(function() return c.Function end)
                if f and type(fn) == "function" then
                    local i, info = pcall(getinfo, fn)
                    if i then
                        local src = tostring(info.short_src or info.source or "")
                        for _, needle in ipairs(needles) do
                            if src:find(needle, 1, true) then
                                if needle == "NotificationController" then return d end
                                best = best or d
                                break
                            end
                        end
                    end
                end
            end
        end
    end
    return best
end

function resolveNotifyRemote()
    if _G.PhiNotifyRemote then return _G.PhiNotifyRemote end
    -- Bounded waits: the previous unbounded WaitForChild could stall everything after
    -- it forever if the game moved the container.
    local Net = ReplicatedStorage:WaitForChild("Packages", 8)
    if Net then Net = Net:WaitForChild("Net", 8) end
    -- The container path moves between builds; if it is gone, scan all of
    -- ReplicatedStorage instead of giving up.
    local found = (Net and scanRootForNotifier(Net)) or scanRootForNotifier(ReplicatedStorage)
    if found then return found end
    return nil
end

function aceStripRich(text)
    if type(text) ~= "string" then return tostring(text) end
    return (text:gsub("<[^>]->", ""))
end

function aceTokenize(text)
    local words = {}
    for word in text:gmatch("[%w_]+") do
        words[#words + 1] = word
    end
    return words
end

-- TRACEDAI RIDDLE SOLVER (FAST + SMART + MULTI-KEY) --
-- "Use AI" riddles are answered with the full TracedAI prompt (embedded game data)
-- and its CODE:/ANSWER:/SEARCHCODE:/SEARCHANSWER: reply format.
-- Speed tricks:
--   1) Answer cache: repeated riddles answer instantly (memory + saved to file).
--   2) Parallel race: every model is asked AT THE SAME TIME; first valid answer wins.
--   3) Hard timeout so a hung request can never stall the next riddle.
-- Quota tricks for free keys:
--   4) Rotating key pool: paste several free keys (comma-separated) in the Tools
--      panel and requests are spread across them, multiplying your rate limit.
--   5) A key that returns 429/quota-exceeded is benched ~15s and another is used.
-- Scoped in a do-end block to stay under Luau's 200-local limit for the main chunk.
local askGeminiRiddle
do
    -- Built-in default key, used automatically when no keys are pasted in Tools.
    -- Auto-routed by prefix: "sk-..." -> OpenAI, anything else -> Gemini.
    -- Built-in pool, used when the Tools box is empty. Rotated like user keys.
    DEFAULT_KEYS = {
        "AQ.Ab8RN6Ju4yUHvNEiKltitdZtJrt_BTpAIydZuCVI7cX0QKZhRQ",
        "AQ.Ab8RN6LOulw1y_hfug38Xd7kS3LjO1eIngqEyM7wtGvGmhTQIA",
        "AQ.Ab8RN6LqesyUvkqqO8KNW-OHC4ZozakjN1w64ywS2PR9ZacjAQ",
    }
    DEFAULT_KEY = DEFAULT_KEYS[1]
    GEMINI_URL_PREFIX = "https://generativelanguage.googleapis.com/v1beta/models/"

    -- Models raced in parallel, first valid answer wins. Pinned versions are used
    -- (not the "latest" alias) because the alias points at the newest and most
    -- demand-spiked model. 2.5-flash is the smartest Gemini model; flash-lite is
    -- fastest. The solver auto-detects which provider each key belongs to:
    --   "sk-..."   -> OpenAI   (uses OPENAI_MODELS)
    --   "AIza..."  -> Gemini   (uses GEMINI_MODELS)
    -- Order matters for the self-test (it probes [1]) and for the race: the first
    -- entry is the fastest non-thinking model, which is what a timed riddle wants.
    GEMINI_MODELS_BASE = {
        "gemini-3.1-flash-lite",
        "gemini-3.5-flash",
        "gemini-3.6-flash",
    }
    GEMINI_MODELS = {}

    -- Google retires models for existing accounts too, and its 404 body names the
    -- replacement, so adopt it and remember it instead of failing forever.
    function rebuildGeminiModels()
        GEMINI_MODELS = {}
        local seen = {}
        for _, name in ipairs(parseModelList(savedConfig.geminiModels)) do
            if not seen[name] then
                seen[name] = true
                GEMINI_MODELS[#GEMINI_MODELS + 1] = name
            end
        end
        for _, name in ipairs(GEMINI_MODELS_BASE) do
            if not seen[name] then
                seen[name] = true
                GEMINI_MODELS[#GEMINI_MODELS + 1] = name
            end
        end
        return GEMINI_MODELS
    end

    function parseModelList(raw)
        local out = {}
        if type(raw) == "string" then
            for chunk in raw:gmatch("[^,;%s]+") do
                local n = chunk:gsub("^models/", ""):gsub("[%s,]", "")
                if n ~= "" then out[#out + 1] = n end
            end
        end
        return out
    end

    -- Pull the replacement model name out of a retirement error.
    function adoptReplacementModel(body)
        local text = type(body) == "string" and body or (type(body) == "table" and tostring(body.Message or body.message) or "")
        if text == "" then return nil end
        local name = text:match("use models/([%w%.%-]+)") or text:match("models/([%w%.%-]+) is no longer available")
        if not name or name == "" then return nil end
        for _, existing in ipairs(GEMINI_MODELS) do
            if existing == name then return nil end
        end
        local kept = {}
        for _, n in ipairs(GEMINI_MODELS) do
            if n ~= name then kept[#kept + 1] = n end
        end
        table.insert(kept, 1, name)
        savedConfig.geminiModels = table.concat(kept, ",")
        rebuildGeminiModels()
        pcall(saveConfig)
        return name
    end

    rebuildGeminiModels()
    OPENAI_MODELS = {
        "gpt-4o-mini",
        "gpt-4.1-mini",
        "gpt-4.1-nano",
    }

    -- Seconds to wait for an answer before giving up (long enough for a busy model).
    RACE_TIMEOUT = 12

    -- Seconds a key is benched after a 429 / quota-exceeded response.
    KEY_COOLDOWN = 15

    aceHttpRequest = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request

    -- === ROTATING KEY POOL ===
    aiKeys = {}
    keyIndex = 0
    keyCooldownUntil = {}

    -- Strip the junk that comes along when a key is copied out of a website, a
    -- chat message or a JSON blob: quotes, backticks, "Bearer ", label prefixes.
    function cleanKeyToken(token)
        local k = tostring(token or "")
        k = k:gsub("^[Bb][Ee][Aa][Rr][Ee][Rr]%s+", "")
        k = k:gsub("^[xX]%-[gG][oO][oO][gG]%-[aA][pP][iI]%-[kK][eE][yY]%s*[:=]%s*", "")
        k = k:gsub("^[kK][eE][yY]%s*[:=]%s*", "")
        k = k:gsub("[\"'`%,;]", "")
        k = k:gsub("^%s+", ""):gsub("%s+$", "")
        k = k:gsub("[\r\n\t]", "")
        return k
    end

    -- A usable key is a long unbroken run of key-ish characters.
    function looksLikeKey(k)
        if type(k) ~= "string" or #k < 20 then return false end
        if k:find("%s") then return false end
        return not not k:match("^[%w%-_%.%:%/%+%=%-]+$")
    end

    -- Which API does this key belong to? Everything we do not recognise is sent
    -- to Gemini, so say so instead of failing mysteriously.
    function classifyKey(key)
        if type(key) ~= "string" then return "junk", "not a string" end
        -- NB: Lua patterns have no {n,} quantifier, so length is checked by hand.
        if key:match("^AIza[%w%-_]+$") and #key >= 30 then return "gemini", "Google API key" end
        -- New AI Studio keys start with "AQ." and work on the same v1beta endpoint.
        if key:match("^AQ%.[%w%-_]+$") and #key >= 40 then return "gemini", "Google API key (AI Studio)" end
        if key:match("^sk%-") then return "openai", "OpenAI key" end
        if key:match("^ya29%.") then return "token", "OAuth access token, not an API key" end
        if key:match("proj%-") or key:match("^%-*sk") then return "openai", "OpenAI key" end
        if #key < 20 then return "junk", "too short (" .. #key .. " chars)" end
        return "unknown", "unrecognised format - tried as a Google key"
    end

    -- Accepted plus rejected tokens, so the Tools row can tell you what it dropped.
    lastKeyParse = { accepted = 0, rejected = 0, notes = {} }

    -- Commas, semicolons, newlines, tabs or plain spaces all separate keys. The old
    -- comma-only split turned a pasted list of keys into one invalid token.
    function parseKeys(raw)
        local keys, seen = {}, {}
        lastKeyParse = { accepted = 0, rejected = 0, notes = {} }
        if type(raw) == "string" and raw ~= "" then
            for chunk in raw:gmatch("[^,;%s]+") do
                local k = cleanKeyToken(chunk)
                if k ~= "" then
                    if looksLikeKey(k) then
                        if not seen[k] then
                            seen[k] = true
                            keys[#keys + 1] = k
                        end
                    else
                        lastKeyParse.rejected = lastKeyParse.rejected + 1
                        local kind, note = classifyKey(k)
                        lastKeyParse.notes[#lastKeyParse.notes + 1] =
                            "dropped [" .. maskKey(k) .. "]: " .. note
                                .. " - paste one key per line, or clear this box to use the built-in pool"
                    end
                end
            end
        end
        lastKeyParse.accepted = #keys
        return keys
    end

    -- Never print a whole key.
    function maskKey(key)
        if type(key) ~= "string" or key == "" then return "<empty>" end
        if #key < 12 then return key .. " (" .. #key .. " chars)" end
        return key:sub(1, 4) .. "..." .. key:sub(-4) .. " (" .. #key .. " chars)"
    end

    function rebuildKeys()
        aiKeys = parseKeys(savedConfig.apiKeys)
        local usingDefault = false
        if #aiKeys == 0 then
            for _, k in ipairs(DEFAULT_KEYS) do
                if looksLikeKey(k) then aiKeys[#aiKeys + 1] = k end
            end
            usingDefault = true
        end
        keyIndex = 0
        aiKeysAreDefault = usingDefault
        return usingDefault
    end
    rebuildKeys()

    -- Called by the Tools-panel API-key box when the keys are edited.
    setAiKeys = function() rebuildKeys() end
    getAiKeyCount = function() return #aiKeys end

    function markKeyBusy(key)
        keyCooldownUntil[key] = os.clock() + KEY_COOLDOWN
    end

    function pickKeyFrom(list)
        for _ = 1, #list do
            keyIndex = keyIndex % #list + 1
            local k = list[keyIndex]
            local untilT = keyCooldownUntil[k]
            if not untilT or os.clock() >= untilT then
                return k
            end
        end
        return nil
    end

    function isRateLimited(code, bodyText)
        if code == 429 then return true end
        if type(bodyText) == "string" then
            local lower = bodyText:lower()
            if lower:find("quota", 1, true)
                or lower:find("rate limit", 1, true)
                or lower:find("rate_limit", 1, true)
                or lower:find("too many requests", 1, true)
                or lower:find("exhausted", 1, true)
                or lower:find("credit", 1, true)
                or lower:find("insufficient", 1, true) then
                return true
            end
        end
        return false
    end

    function isOpenAIKey(key)
        return classifyKey(key) == "openai"
    end

    function requestModel(modelName, key, promptText, question)
        if isOpenAIKey(key) then
            local body = HttpService:JSONEncode({
                model = modelName,
                messages = {
                    { role = "system", content = promptText },
                    { role = "user", content = question },
                },
                temperature = 0.1,
                max_completion_tokens = 512,
            })
            return aceHttpRequest({
                Url = "https://api.openai.com/v1/chat/completions",
                Method = "POST",
                Headers = {
                    ["Content-Type"] = "application/json",
                    ["Authorization"] = "Bearer " .. key,
                },
                Body = body,
            })
        end
        local body = HttpService:JSONEncode({
            systemInstruction = { parts = {{ text = promptText }} },
            contents = { { role = "user", parts = {{ text = question }} } },
            -- 3.5/3.6 spend tokens on hidden thinking first; a small cap comes back
            -- as candidates[].content with no parts, which looks like "model failed".
            generationConfig = { temperature = 0.1, maxOutputTokens = 1024 },
        })
        return aceHttpRequest({
            Url = GEMINI_URL_PREFIX .. modelName .. ":generateContent",
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json",
                ["x-goog-api-key"] = key,
            },
            Body = body,
        })

    function extractText(response, openAI)
        local raw = response and response.Body
        local decoded = raw
        if type(raw) == "string" then
            -- Some executors return the body already decoded as a table; handle both.
            local ok, d = pcall(HttpService.JSONDecode, HttpService, raw)
            if not ok or type(d) ~= "table" then return nil end
            decoded = d
        elseif type(raw) ~= "table" then
            return nil
        end
        if openAI then
            local choice = decoded.choices and decoded.choices[1]
            local msg = choice and choice.message
            if msg and type(msg.content) == "string" and msg.content ~= "" then
                return msg.content
            end
            return nil
        end
        local candidate = decoded.candidates and decoded.candidates[1]
        local parts = candidate and candidate.content and candidate.content.parts
        if type(parts) ~= "table" then return nil end
        -- Skip "thought" parts some Gemini models emit; take the first real text part.
        for _, part in ipairs(parts) do
            if type(part.text) == "string" and part.text ~= "" and not part.thought then
                return part.text
            end
        end
        return nil
    end

    MONTH_NAMES = { "January", "February", "March", "April", "May", "June",

    function buildLiveState()
        local ok, t = pcall(os.date, "*t")
        if not ok or not t then
            -- Some executors restrict os.date("*t"); fall back to formatted strings.
            local ok2, today = pcall(os.date, "%d %B %Y")
            local _, month = pcall(os.date, "%B")
            if ok2 and today then
                return "- Today's date: " .. today .. ".\n- Current month: " .. (month or "unknown") .. "."
            end
            return "- Today's date is not available in this executor."
        end
        local monthName = MONTH_NAMES[t.month] or ("Month " .. tostring(t.month))
        return table.concat({
            "- Today's date: " .. tostring(t.day) .. " " .. monthName .. " " .. tostring(t.year)
                .. " (day " .. tostring(t.day) .. ", month " .. tostring(t.month) .. ", year " .. tostring(t.year) .. ").",
            "- Current month: " .. monthName .. " (month number " .. tostring(t.month) .. ").",
            "- Current year: " .. tostring(t.year) .. ".",
        }, "\n")
    end

    function buildPrompt()
        local facts = {}
        if _useRiddleDB then
            for dbQuestion, dbAnswer in pairs(riddleDB) do
                facts[#facts + 1] = dbQuestion .. " = " .. dbAnswer
                if #facts >= 120 then break end
            end
        end
        local extra = (#facts > 0)
            and ("\n\n=== EXTRA KNOWN FACTS (question = answer) ===\n" .. table.concat(facts, "\n"))
            or ""
        return PROMPT_BODY
            .. "\n\n=== DEVELOPER ===\n" .. DEVELOPER
            .. "\n\n=== LIVE STATE ===\n" .. buildLiveState()
            .. extra
            .. "\n\n=== GAME DATA ===\n" .. GAME_DATA
    end

    -- Turn the model's tagged reply into the answer to paste.
    function parseTracedAnswer(raw)
        local text = trimRiddleAnswer(raw)
        if text == "" then return nil end
        local upper = text:upper()
        local mode, payload
        if upper:sub(1, 11) == "SEARCHCODE:" then
            mode, payload = "CODE", text:sub(12)
            local pipe = payload:find(" | ", 1, true)
            if pipe then payload = payload:sub(pipe + 3) end
        elseif upper:sub(1, 13) == "SEARCHANSWER:" then
            mode, payload = "ANSWER", text:sub(14)
            local pipe = payload:find(" | ", 1, true)
            if pipe then payload = payload:sub(pipe + 3) end
        elseif upper:sub(1, 5) == "CODE:" then
            mode, payload = "CODE", text:sub(6)
        elseif upper:sub(1, 7) == "ANSWER:" then
            mode, payload = "ANSWER", text:sub(8)
        else
            mode, payload = "CODE", text
        end
        local alt = payload:find("|| ALT:", 1, true)
        if alt then payload = payload:sub(1, alt - 1) end
        payload = trimRiddleAnswer(payload)
        if payload == "" then return nil end
        if mode == "CODE" then
            payload = payload:upper()
            payload = payload:gsub("[^%w]", "")
        else
            payload = payload:gsub(",", "")
        end
        return payload
    end

    askGeminiRiddle = function(question)
        if aiBusy then return end
        local seenKey = normalizeRiddleQuestion(question)
        if seenKey == "" or aiSeen[seenKey] then return end
        aiSeen[seenKey] = true
        task.delay(30, function() aiSeen[seenKey] = nil end)

        -- Cache hit: answer instantly, no network at all.
        if aiCache[seenKey] then
            setStatus("AI (cached): " .. aiCache[seenKey], COLORS.Green)
            submitRiddleAnswer(aiCache[seenKey], "AI")
            return
        end

        if type(aceHttpRequest) ~= "function" then
            setStatus("AI: executor has no HTTP request function", COLORS.Red)
            return
        end

        aiBusy = true
        setStatus("AI: thinking...", COLORS.Dim)
        task.spawn(function()
            local promptText = buildPrompt()

            -- Partition keys by provider, then race each provider's models.
            -- Auto-detection: "sk-..." -> OpenAI, everything else -> Gemini.
            local openaiKeys = {}
            local geminiKeys = {}
            for _, k in ipairs(aiKeys) do
                if isOpenAIKey(k) then openaiKeys[#openaiKeys + 1] = k
                else geminiKeys[#geminiKeys + 1] = k end
            end

            local pending = 0
            if #openaiKeys > 0 then pending = pending + #OPENAI_MODELS end
            if #geminiKeys > 0 then pending = pending + #GEMINI_MODELS end
            if pending == 0 then
                aiBusy = false
                setStatus("AI: no API key configured", COLORS.Red)
                return
            end

            local done = false
            local settled = 0
            local failReason = ""

            local function resolve(answer, modelName)
                if done then return end
                done = true
                aiBusy = false
                if answer and answer ~= "" then
                    aiCache[seenKey] = answer
                    saveAiCache()
                    setStatus("AI solved [" .. tostring(modelName) .. "]: " .. answer, COLORS.Green)
                    submitRiddleAnswer(answer, "AI")
                end
            end

            local function settle()
                settled = settled + 1
                if settled >= pending and not done then
                    resolve(nil, nil)
                    local why = failReason ~= "" and failReason or "no answer returned"
                    setStatus("AI: all models failed (" .. why .. ")", COLORS.Red)
                end
            end

            -- Race every model of a provider in parallel; the first valid answer
            -- wins. Each worker rotates through that provider's keys, benching a
            -- rate-limited key and moving on to the next.
            local function raceProvider(models, keys, isOpenAI)
                if #keys == 0 then return end -- never spawn phantom workers
                for _, modelName in ipairs(models) do
                    task.spawn(function()
                        for _ = 1, #keys do
                            if done then return end
                            local key = pickKeyFrom(keys)
                            if not key then break end
                            local ok, response = pcall(requestModel, modelName, key, promptText, question)
                            if not ok then
                                failReason = "request error (" .. tostring(response) .. ")"
                            else
                                local code = tonumber(response and (response.StatusCode or response.Status or response.code)) or 0
                                if code >= 200 and code < 300 then
                                    local ok2, text = pcall(extractText, response, isOpenAI)
                                    if ok2 and text then
                                        local answer = parseTracedAnswer(text)
                                        if answer and answer ~= "" and answer:upper() ~= "SKIP" then
                                            resolve(answer, modelName)
                                            return
                                        end
                                    end
                                    failReason = "model returned no usable answer"
                                    break -- 200 but no usable answer: stop this model
                                elseif isRateLimited(code, response.Body) then
                                    failReason = "rate limited (HTTP " .. tostring(code) .. ")"
                                    markKeyBusy(key)
                                    -- rotate to the next key and retry this model
                                elseif code == 404 then
                                    local adopted = adoptReplacementModel(response.Body)
                                    if adopted then
                                        failReason = "model retired - now using " .. adopted
                                        if type(warn) == "function" then
                                            warn("[CALAMITY SNIPER] Google retired a model; switching to " .. adopted)
                                        end
                                    else
                                        failReason = "HTTP 404 - model not available for this key"
                                    end
                                    break -- the list is fixed for the next attempt
                                else
                                    failReason = "HTTP " .. tostring(code) .. " - key rejected"
                                    break -- auth / not-found: this model won't work here
                                end
                            end
                        end
                        if not done then settle() end
                    end)
                end
            end

            raceProvider(OPENAI_MODELS, openaiKeys, true)
            raceProvider(GEMINI_MODELS, geminiKeys, false)

            -- Hard timeout: never leave the solver stuck waiting on a slow model.
            task.delay(RACE_TIMEOUT, function()
                if not done then
                    resolve(nil, nil)
                    setStatus("AI: timed out after " .. tostring(RACE_TIMEOUT) .. "s", COLORS.Red)
                end
            end)
        end)
    end
end

-- Turn a bare status code into something actionable.
function describeApiError(code, msg)
    msg = tostring(msg or "")
    local lower = msg:lower()
    if code == 404 then
        return "model retired for this account - the script adopts Google's suggested replacement automatically"
    end
    if code == 400 then
        if lower:find("api key not valid", 1, true) or lower:find("invalid api key", 1, true) or lower == "" then
            return "that string is not a Google API key - copy it from aistudio.google.com/apikey (AIza... and AQ... both work)"
        end
        if lower:find("permission", 1, true) then return "key refused for this project/API" end
    elseif code == 401 then
        if lower:find("access token", 1, true) or lower:find("access_token", 1, true)
            or lower:find("token type", 1, true) or lower:find("token_type", 1, true)
            or lower:find("unsupported", 1, true) then
            return "key refused - a revoked/expired AI Studio key, or an OAuth access token (ya29...) pasted instead of an API key"
        end
        return "key rejected (expired, revoked or typed wrong)"
    elseif code == 403 then
        return "key is valid but not allowed here - enable the Generative Language API for its project"
    elseif code == 404 then
        return "endpoint/model not found for this key"
    elseif code == 429 then
        return "quota exhausted for this key (the free tier is very tight on flash models)"
    end
    if code >= 500 then return "provider error - try again in a moment" end
    return msg ~= "" and msg or ("HTTP " .. tostring(code))
end

function apiErrorText(response)
    local raw = response and response.Body
    local decoded = raw
    if type(raw) == "string" then
        local ok, d = pcall(HttpService.JSONDecode, HttpService, raw)
        if not ok or type(d) ~= "table" then return "" end
        decoded = d
    end
    if type(decoded) ~= "table" then return "" end
    local err = decoded.error
    if type(err) ~= "table" then return "" end
    local msg = err.message or err.status
    if type(msg) ~= "string" then return "" end
    msg = msg:gsub("\n", " "):gsub("^%s+", ""):gsub("%s+$", "")
    return #msg > 120 and msg:sub(1, 120) .. "..." or msg
end

                            .. string.format("%.1f", os.clock() - started) .. "s", COLORS.Green)
                        report(true, tag .. " works (" .. answer .. ")", COLORS.Green)
                        return
                    end
                    failures[#failures + 1] = tag .. ": 200 but no text part"
                else
                    local why = apiErrorText(response)
                    if code == 404 then
                        local adopted = adoptReplacementModel(response.Body)
                        if adopted then
                            consoleNote("ai-test", "model retired by Google - pool is now "
                                .. table.concat(GEMINI_MODELS, ", "), COLORS.Green)
                        end
                    end
                    failures[#failures + 1] = tag .. " " .. maskKey(key) .. ": HTTP "
                        .. tostring(code) .. " - " .. describeApiError(code, why)
                end
            end
        end
        aiBusy = false
        for _, line in ipairs(failures) do consoleNote("ai-test", line, COLORS.Red) end
        report(false, "every key failed (" .. #failures .. " tried)", COLORS.Red)
    end)
end

aceCollectBuffer = {}

-- Detect full codes embedded in ordinary announcements such as:
-- "The code is ABcdefghijklmno" or "The code was ABcdefghijklmno".
-- These are complete codes, not riddle fragments, so they bypass the 1-5 message counter.
function detectAnnouncementCode(text)
    if type(text) ~= "string" then return nil end
    local patterns = {
        "[Tt]he%s+code%s+is%s*[:%-]?%s*[\"']?([%w_%-]+)",
        "[Tt]he%s+code%s+was%s*[:%-]?%s*[\"']?([%w_%-]+)",
        "[Cc]ode%s*[:%-]%s*[\"']?([%w_%-]+)",
    }
    for _, pattern in ipairs(patterns) do
        local found = text:match(pattern)
        if found and #found >= 3 then
            found = found:gsub("[^%w_]", "")
            if _caseMode == "UPPER" then found = found:upper()
            elseif _caseMode == "lower" then found = found:lower() end
            return found
        end
    end
    return nil
end

function onAceAnnouncement(...)
    local text = aceStripRich(tostring((...) or ""))
    text = text:match("^%s*(.-)%s*$") or ""
    local announcedCode = detectAnnouncementCode(text)
    if announcedCode then
        -- Direct full-code announcement: log, paste, and submit immediately when Auto submit is enabled.
        setStatus("Detected code: " .. announcedCode, COLORS.Green)
        submitRiddleAnswer(announcedCode, "Detected code")
        return
    end
    local riddleAnswer = answerRiddle(text)
    if riddleAnswer then
        setStatus("Riddle solved: " .. riddleAnswer, COLORS.Green)
        submitRiddleAnswer(riddleAnswer, "Riddle")
        return
    end
    if text == "" then return end
    if text:find("%s") then
        -- Multi-word message the database couldn't solve: hand it to Gemini when Use AI is ON.
        if _useAI and _riddleSolver then askGeminiRiddle(text) end
        return
    end
    
    for _, word in ipairs(aceTokenize(text)) do
        aceCollectBuffer[#aceCollectBuffer + 1] = word
    end
    
    local parts = {}
    for index = 1, math.min(#aceCollectBuffer, ACE_WORD_COUNT) do
        parts[index] = aceCollectBuffer[index]
    end
    if #aceCollectBuffer < ACE_WORD_COUNT then return end
    aceCollectBuffer = {}
    
    local captured = table.concat(parts)
    if _caseMode == "UPPER" then captured = captured:upper() elseif _caseMode == "lower" then captured = captured:lower() end
    if _blacklist ~= "" then for term in _blacklist:lower():gmatch("[^,]+") do if captured:lower():find(term:match("^%s*(.-)%s*$"), 1, true) then return end end end
    if captured == "" or _seen[captured] then return end
    _seen[captured] = true
    task.delay(1.25, function() _seen[captured] = nil end)
    appendToBox(captured)
end

local aceRemoteOk, aceRemoteFound = pcall(resolveNotifyRemote)
aceNotifyRemote = (aceRemoteOk and aceRemoteFound) or nil
local aceListenConnection
if aceNotifyRemote then
    if getgenv then
        local previous = getgenv().ACECodeSniperNotifyConnection
        if previous then pcall(function() previous:Disconnect() end) end
    end
    aceListenConnection = aceNotifyRemote.OnClientEvent:Connect(function(...)
        if not _enabled then return end
        pcall(onAceAnnouncement, ...)
    end)
    if getgenv then getgenv().ACECodeSniperNotifyConnection = aceListenConnection end
else
    -- Best effort: public chat still carries codes and riddles that are typed or
    -- reposted, so hook it rather than doing nothing. This is NOT the game's own
    -- notification feed, so private announcement-only riddles stay unheard.
    local chatHooked = false
    pcall(function()
        local TextChatService = game:GetService("TextChatService")
        if TextChatService and TextChatService.OnClientMessageReceived then
            aceChatConnection = TextChatService.OnClientMessageReceived:Connect(function(message)
                if not _enabled then return end
                local okText, text = pcall(function() return message.Text end)
                if okText and type(text) == "string" and text ~= "" then
                    pcall(onAceAnnouncement, text)
                end
            end)
            chatHooked = true
        end
    end)
    if not chatHooked then
        pcall(function()
            local TextChat = game:GetService("TextChat")
            if TextChat and TextChat.TextReceived then
                aceChatConnection = TextChat.TextReceived:Connect(function(text)
                    if _enabled and type(text) == "string" then pcall(onAceAnnouncement, text) end
                end)
                chatHooked = true
            end
        end)
    end
    if type(warn) == "function" then
        warn("[CALAMITY SNIPER] notify remote not found"
            .. (notifyScanNote and (" - " .. notifyScanNote) or "") .. " - "
            .. (chatHooked and "listening to public chat only (announcement-only riddles will be missed)"
                or "codes, riddles and AI stay silent"))
    end
    setStatus(chatHooked and "input: chat only" or "riddle input: NOT FOUND",
        chatHooked and COLORS.Dim or COLORS.Red)
end
